#include <stdio.h>
#include <stdlib.h>

struct Node {
    int vertex;
    struct Node* next;
};

struct Node* adj[5];

void addEdge(int u, int v) {
    struct Node* newNode = malloc(sizeof(struct Node));
    newNode->vertex = v;
    newNode->next = adj[u];
    adj[u] = newNode;
}

void display() {
    for(int i = 0; i < 5; i++) {
        struct Node* temp = adj[i];
        printf("Vertex %d: ", i);
        while(temp) {
            printf("%d -> ", temp->vertex);
            temp = temp->next;
        }
        printf("NULL\n");
    }
}

int main() {
    for(int i = 0; i < 5; i++)
        adj[i] = NULL;

    addEdge(0,1);
    addEdge(0,2);
    addEdge(1,3);
    addEdge(3,4);

    display();

    return 0;
}
